cmake_minimum_required(VERSION 3.20)
project(mozilla_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(MOZILLA_SOURCE_REVISION "8eada038fd45bbf6fa58a412b22f80c79887fabc")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "" FORCE)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_compile_options(-fexceptions -ffunction-sections -fdata-sections)
add_compile_definitions(EIGEN_MPL2_ONLY)
include_directories("${CMAKE_CURRENT_SOURCE_DIR}")
set(GIT_SUBMODULE OFF CACHE BOOL "" FORCE)
set(USE_WASM_COMPATIBLE_SOURCE ON CACHE BOOL "" FORCE)
set(COMPILE_WITHOUT_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(USE_THREADS ON CACHE BOOL "" FORCE)
set(USE_MKL OFF CACHE BOOL "" FORCE)
set(USE_ONNX_SGEMM ON CACHE BOOL "" FORCE)
set(USE_DOXYGEN OFF CACHE BOOL "" FORCE)
set(COMPILE_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_TESTS OFF CACHE BOOL "" FORCE)
set(COMPILE_LIBRARY_ONLY ON CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(RUY_MINIMAL_BUILD ON CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_UNIT_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_MOCK_TESTS OFF CACHE BOOL "" FORCE)
set(CPUINFO_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(CLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2GREP OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_8 ON CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_16 OFF CACHE BOOL "" FORCE)
set(PCRE2_BUILD_PCRE2_32 OFF CACHE BOOL "" FORCE)
set(PCRE2_SUPPORT_JIT OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  set(BUILD_ARCH armv8-a CACHE STRING "" FORCE)
  set(USE_RUY ON CACHE BOOL "" FORCE)
  set(USE_RUY_SGEMM OFF CACHE BOOL "" FORCE)
  set(USE_INTGEMM OFF CACHE BOOL "" FORCE)
  # Pinned SIMD_Utils uses FE_* before declaring its fenv prerequisites.
  add_compile_options(-include fenv.h)
elseif(ANDROID_ABI STREQUAL "x86_64")
  set(BUILD_ARCH x86-64 CACHE STRING "" FORCE)
  set(USE_RUY OFF CACHE BOOL "" FORCE)
  set(USE_RUY_SGEMM OFF CACHE BOOL "" FORCE)
else()
  message(FATAL_ERROR "Only arm64-v8a and x86_64 are supported")
endif()
add_subdirectory("${MOZILLA_PCRE_SOURCE}" pcre2 EXCLUDE_FROM_ALL)
set(PCRE2_LIBRARIES pcre2-8-static CACHE STRING "" FORCE)
set(PCRE2_INCLUDE_DIRS "${CMAKE_CURRENT_BINARY_DIR}/pcre2" CACHE STRING "" FORCE)
add_subdirectory("${MOZILLA_SOURCE}/inference" inference EXCLUDE_FROM_ALL)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  # Marian's SIMD_Utils SSE selector collides with Eigen's SSE enum.
  # ONNX SGEMM uses Eigen's native architecture detection, not that selector.
  target_compile_options(onnx-sgemm PRIVATE -USSE)
  # Upstream exposes Marian ARM headers but does not propagate their selectors.
  target_compile_definitions(bergamot-translator-source PUBLIC ARM FMA SSE)
  target_compile_options(bergamot-translator-source PUBLIC -flax-vector-conversions -fno-strict-aliasing)
endif()
add_library(mozilla_jni SHARED mozilla_jni.cpp)
target_link_libraries(mozilla_jni PRIVATE bergamot-translator-source)
target_link_options(mozilla_jni PRIVATE
  -Wl,--no-undefined -Wl,--gc-sections -Wl,--exclude-libs,ALL
  "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
  -Wl,-z,max-page-size=16384 -Wl,-z,common-page-size=16384)
set_target_properties(mozilla_jni PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/jni")
get_directory_property(marian_arch DIRECTORY "${MOZILLA_SOURCE}/inference/marian-fork" DEFINITION CMAKE_TARGET_ARCHITECTURE_CODE)
get_directory_property(marian_intgemm DIRECTORY "${MOZILLA_SOURCE}/inference/marian-fork" DEFINITION USE_INTGEMM)
get_directory_property(marian_ruy DIRECTORY "${MOZILLA_SOURCE}/inference/marian-fork" DEFINITION USE_RUY)
if(ANDROID_ABI STREQUAL "arm64-v8a")
  if(NOT marian_arch MATCHES "arm" OR NOT marian_ruy OR marian_intgemm)
    message(FATAL_ERROR "ARM must dispatch to the pinned Ruy/NEON backend")
  endif()
elseif(NOT marian_intgemm OR marian_ruy)
  message(FATAL_ERROR "x86_64 must use the baseline intgemm backend")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/abi-settings.txt"
  "architecture=${marian_arch}\nintgemm=${marian_intgemm}\nruy=${marian_ruy}\n")
